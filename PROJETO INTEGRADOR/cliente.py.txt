import socket

HOST = '127.0.0.1'
PORTA = 5000


# Envia uma mensagem
def enviar_mensagem(conexao, mensagem):
    dados = mensagem.encode('utf-8')
    conexao.sendall(len(dados).to_bytes(4, byteorder='big'))
    conexao.sendall(dados)


# Recebe a quantidade de bytes necessária
def receber_bytes(conexao, quantidade):
    dados = b''

    while len(dados) < quantidade:
        parte = conexao.recv(quantidade - len(dados))

        if not parte:
            return None

        dados += parte

    return dados


# Recebe uma mensagem
def receber_mensagem(conexao):
    tamanho_bytes = receber_bytes(conexao, 4)

    if not tamanho_bytes:
        return None

    tamanho = int.from_bytes(tamanho_bytes, byteorder='big')
    dados = receber_bytes(conexao, tamanho)

    if not dados:
        return None

    return dados.decode('utf-8')


def cliente():
    # Cria o socket TCP
    cliente_socket = socket.socket(
        socket.AF_INET,
        socket.SOCK_STREAM
    )

    try:
        cliente_socket.connect((HOST, PORTA))

        print("\n" + "=" * 50)
        print("          CAIXA ELETRÔNICO")
        print("=" * 50)

        autenticado = False

        # Login
        while not autenticado:
            login = input("Digite seu login: ")
            senha = input("Digite sua senha: ")

            enviar_mensagem(cliente_socket, login)
            enviar_mensagem(cliente_socket, senha)

            resposta = receber_mensagem(cliente_socket)

            if resposta is None:
                print("[CLIENTE] Servidor encerrou a conexão.")
                return

            if resposta == "sucesso":
                print("\n[CLIENTE] Login realizado com sucesso!")
                autenticado = True
            else:
                print("\n[CLIENTE] Login ou senha inválidos.")
                print("[CLIENTE] Tente novamente.")

        # Menu
        while True:
            print("\n" + "=" * 50)
            print("          MENU DE OPERAÇÕES")
            print("=" * 50)
            print("1 - Depositar")
            print("2 - Sacar")
            print("3 - Visualizar Saldo")
            print("4 - Sair")
            print("=" * 50)

            opcao = input("Escolha uma opção: ")

            # Depósito
            if opcao == '1':
                valor = input("Digite o valor do depósito: ")

                enviar_mensagem(cliente_socket, opcao)
                enviar_mensagem(cliente_socket, valor)

                resposta = receber_mensagem(cliente_socket)

                print(f"\n[CLIENTE] Servidor: {resposta}")

            # Saque
            elif opcao == '2':
                valor = input("Digite o valor do saque: ")

                enviar_mensagem(cliente_socket, opcao)
                enviar_mensagem(cliente_socket, valor)

                resposta = receber_mensagem(cliente_socket)

                print(f"\n[CLIENTE] Servidor: {resposta}")

            # Consulta do saldo
            elif opcao == '3':
                enviar_mensagem(cliente_socket, opcao)

                resposta = receber_mensagem(cliente_socket)

                print(f"\n[CLIENTE] Servidor: {resposta}")

            # Sair
            elif opcao == '4':
                enviar_mensagem(cliente_socket, opcao)

                resposta = receber_mensagem(cliente_socket)

                print(f"\n[CLIENTE] Servidor: {resposta}")

                break

            else:
                print("\n[CLIENTE] Opção inválida.")
                print("[CLIENTE] Escolha uma opção de 1 a 4.")

    except ConnectionRefusedError:
        print("[CLIENTE] Não foi possível conectar ao servidor.")

    except ConnectionResetError:
        print("[CLIENTE] Conexão interrompida pelo servidor.")

    finally:
        cliente_socket.close()
        print("[CLIENTE] Conexão encerrada.")


if __name__ == "__main__":
    cliente()

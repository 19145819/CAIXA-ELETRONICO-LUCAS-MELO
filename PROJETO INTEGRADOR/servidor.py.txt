import socket

HOST = '127.0.0.1'
PORTA = 5000

USUARIO_TESTE = 'USUARIO TESTES'
SENHA_TESTE = '54321'

saldo = 0.0


# Envia uma mensagem
def enviar_mensagem(conexao, mensagem):
    dados = mensagem.encode('utf-8')
    conexao.sendall(len(dados).to_bytes(4, byteorder='big'))
    conexao.sendall(dados)


# Recebe a quantidade de bytes necessária
def receber_bytes(conexao, quantidade):
    dados = b''

    while len(dados) < quantidade:
        parte = conexao.recv(quantidade - len(dados))

        if not parte:
            return None

        dados += parte

    return dados


# Recebe uma mensagem
def receber_mensagem(conexao):
    tamanho_bytes = receber_bytes(conexao, 4)

    if not tamanho_bytes:
        return None

    tamanho = int.from_bytes(tamanho_bytes, byteorder='big')
    dados = receber_bytes(conexao, tamanho)

    if not dados:
        return None

    return dados.decode('utf-8')


def servidor():
    global saldo

    # Cria o socket TCP
    servidor_socket = socket.socket(
        socket.AF_INET,
        socket.SOCK_STREAM
    )

    servidor_socket.setsockopt(
        socket.SOL_SOCKET,
        socket.SO_REUSEADDR,
        1
    )

    servidor_socket.bind((HOST, PORTA))
    servidor_socket.listen(1)

    print("=" * 50)
    print("       SERVIDOR DO CAIXA ELETRÔNICO")
    print("=" * 50)
    print(f"[SERVIDOR] Aguardando conexão em {HOST}:{PORTA}")

    conexao, endereco = servidor_socket.accept()

    print(f"[SERVIDOR] Cliente conectado: {endereco}")

    try:
        autenticado = False

        # Validação do login
        while not autenticado:
            login = receber_mensagem(conexao)

            if login is None:
                return

            senha = receber_mensagem(conexao)

            if senha is None:
                return

            print(f"[SERVIDOR] Tentativa de login: {login}")

            if login == USUARIO_TESTE and senha == SENHA_TESTE:
                enviar_mensagem(conexao, "sucesso")
                autenticado = True
                print("[SERVIDOR] Login realizado com sucesso.")
            else:
                enviar_mensagem(conexao, "falha")
                print("[SERVIDOR] Login ou senha incorretos.")

        # Menu de operações
        while True:
            opcao = receber_mensagem(conexao)

            if opcao is None:
                break

            # Depósito
            if opcao == '1':
                valor_texto = receber_mensagem(conexao)

                if valor_texto is None:
                    break

                try:
                    valor = float(valor_texto.replace(',', '.'))

                    if valor <= 0:
                        enviar_mensagem(
                            conexao,
                            "Valor de depósito inválido."
                        )
                    else:
                        saldo += valor

                        enviar_mensagem(
                            conexao,
                            f"Depósito realizado com sucesso. "
                            f"Novo saldo: R$ {saldo:.2f}"
                        )

                        print(f"[SERVIDOR] Depósito: R$ {valor:.2f}")

                except ValueError:
                    enviar_mensagem(conexao, "Valor inválido.")

            # Saque
            elif opcao == '2':
                valor_texto = receber_mensagem(conexao)

                if valor_texto is None:
                    break

                try:
                    valor = float(valor_texto.replace(',', '.'))

                    if valor <= 0:
                        enviar_mensagem(
                            conexao,
                            "Valor de saque inválido."
                        )
                    elif valor > saldo:
                        enviar_mensagem(
                            conexao,
                            "Saldo insuficiente."
                        )
                    else:
                        saldo -= valor

                        enviar_mensagem(
                            conexao,
                            f"Saque realizado com sucesso. "
                            f"Novo saldo: R$ {saldo:.2f}"
                        )

                        print(f"[SERVIDOR] Saque: R$ {valor:.2f}")

                except ValueError:
                    enviar_mensagem(conexao, "Valor inválido.")

            # Consulta do saldo
            elif opcao == '3':
                enviar_mensagem(
                    conexao,
                    f"Saldo atual: R$ {saldo:.2f}"
                )

                print("[SERVIDOR] Saldo consultado.")

            # Encerramento
            elif opcao == '4':
                enviar_mensagem(
                    conexao,
                    "Conexão encerrada. Até logo!"
                )

                print("[SERVIDOR] Cliente encerrou a conexão.")
                break

            else:
                enviar_mensagem(conexao, "Opção inválida.")

    except ConnectionResetError:
        print("[SERVIDOR] Conexão interrompida.")

    finally:
        conexao.close()
        servidor_socket.close()
        print("[SERVIDOR] Servidor encerrado.")


if __name__ == "__main__":
    servidor()
